<script>
// ── CONFIG ─────────────────────────────────────────────────
// Replace with your actual Gist ID (same as GIST_ID in the .ino)
const GIST_ID = "b50f62eb36a096daf15546c60cfcdd63";
const REFRESH_MS = 30000; // 30 seconds

// ── DOM refs ───────────────────────────────────────────────
const $ = id => document.getElementById(id);
const refreshBar = $('refreshBar');
const errorBanner = $('errorBanner');
const liveDot = $('liveDot');
const liveLabel = $('liveLabel');

// ── State ──────────────────────────────────────────────────
let lastData = null;
let refreshTimer = null;

// ── Fetch & render ─────────────────────────────────────────
async function fetchData() {
// Animate progress bar
refreshBar.style.width = '0%';
refreshBar.style.opacity = '1';
await sleep(50);
refreshBar.style.transition = 'width 1.5s linear';
refreshBar.style.width = '80%';

try {
const res = await fetch(
`https://api.github.com/gists/${GIST_ID}`,
{ headers: { Accept: 'application/vnd.github.v3+json' } }
);
if (!res.ok) throw new Error(`HTTP ${res.status}`);

const gist = await res.json();
const raw = gist.files['esp32_data.json']?.content;
if (!raw) throw new Error('esp32_data.json not found in Gist');

const data = JSON.parse(raw);
lastData = data;
errorBanner.classList.remove('show');
render(data);
setLive(true);
} catch (err) {
console.error(err);
setLive(false);
errorBanner.classList.add('show');
} finally {
// Complete bar
refreshBar.style.transition = 'width 0.2s linear';
refreshBar.style.width = '100%';
await sleep(300);
refreshBar.style.opacity = '0';
refreshBar.style.transition = '';
refreshBar.style.width = '0%';
}
}

function render(d) {
const now = new Date().toLocaleTimeString();
$('syncTime').textContent = `Last sync: ${now}`;
$('uptimeDisplay').textContent = `Uptime: ${d.timestamp ?? '—'}`;
$('footerGistId').textContent = `Gist: ${GIST_ID.slice(0, 10)}…`;

// Static LED
setCard('cardStaticLed', 'valStaticLed', d.static_led, 'ON', 'OFF');

// Blinking
const blink = d.is_blinking;
setCard('cardBlink', 'valBlink', blink, 'ACTIVE', 'IDLE');
$('valInterval').textContent = d.blink_interval_ms ?? 50;

// Blink LED indicators (derived from blink state — board alternates them)
setCard('cardBlink1', 'valBlink1', blink, '◉ ON', '◯ OFF', true);
setCard('cardBlink2', 'valBlink2', blink, '◉ ON', '◯ OFF', true);

// Music
const playing = d.is_playing;
setCard('cardMusic', 'valMusic', playing, '▶ PLAYING', '■ STOPPED', false, 'play');
const vol = d.volume ?? 30;
$('valVolume').textContent = vol;
$('volFill').style.width = `${(vol / 30) * 100}%`;

// Switches
setLed('sw1Led', d.switches?.sw1);
setLed('sw2Led', d.switches?.sw2);
setLed('sw3Led', d.switches?.sw3);

// Buttons
setChip('btnCh6', d.buttons?.ch6);
setChip('btnCh3', d.buttons?.ch3);
setChip('btnCh0', d.buttons?.ch0);

const combo = (d.buttons?.ch6 && (d.buttons?.ch3 || d.buttons?.ch0));
$('comboStatus').textContent = combo
? '⚡ ACTIVE — Trigger fired!'
: 'CH6+CH3 or CH6+CH0';
$('comboStatus').style.color = combo ? 'var(--cyan)' : '';
}

function setCard(cardId, valId, active, onText, offText, warn = false, customClass = null) {
const card = $(cardId);
const val = $(valId);
card.className = 'card' + (active ? (warn ? ' warn' : ' active') : '');
val.className = 'card-value ' + (active ? (customClass ?? 'on') : 'off');
val.textContent = active ? onText : offText;
}

function setLed(id, on) {
const el = $(id);
el.className = 'pill-led' + (on ? ' on' : '');
}

function setChip(id, active) {
const el = $(id);
el.className = 'btn-chip' + (active ? ' active' : '');
}

function setLive(online) {
liveDot.className = 'live-dot' + (online ? '' : ' offline');
liveLabel.textContent = online ? 'LIVE' : 'OFFLINE';
liveLabel.style.color = online ? 'var(--cyan)' : 'var(--red)';
}

function sleep(ms) { return new Promise(r => setTimeout(r, ms)); }

// ── Countdown in footer ────────────────────────────────────
let countdown = REFRESH_MS / 1000;
setInterval(() => {
countdown--;
if (countdown <= 0) countdown = REFRESH_MS / 1000;
$('footerRefresh').textContent = `Next refresh in ${countdown}s`;
}, 1000);

// ── Init ───────────────────────────────────────────────────
fetchData();
setInterval(fetchData, REFRESH_MS);
</script>
